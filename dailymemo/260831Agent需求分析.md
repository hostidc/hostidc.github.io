## 🎯 **2026 H2 给独立开发者/ISV 的「生存与突围」策略建议**

### 1.  **「只做 Tier S + Tier A 的标品化模块供应商」**

* **不要**试图覆盖全行业。选 **1-2 个 Tier S/A 赛道**，把 **Agent 编排引擎 + 行业知识图谱/规则库 + 评测数据集 + 合规/审计/计费框架** 做成 **「可复用、可配置、可白标、可计费」的产品包**。
* **交付物标准**：`Docker Image + Helm Chart + OpenAPI Spec + 评测报告 + 运营后台 + 文档/视频`。

### 2.  **「数据资产 > 模型能力 > 提示词工程」**

* 2026 年模型能力趋同，**你的核心护城河是：行业高质量「指令微调数据/偏好对齐数据/黄金评测集/规则库/案例库」**。每个项目必须沉淀数据资产，下一项目复用率 > 60%。

### 3.  **「合规/审计/可解释/可熔断」是入场券，不是加分项**

* **金融/医疗/政务/司法/能源**项目，**无「全链路审计日志/决策溯源/人工介入熔断/数据脱敏/等保三级/备案号」直接不进门**。把这套能力做成 **通用中间件/基础设施层**，每个项目复用。

### 4.  **「端侧/混合部署」是新标配**

* **工业/机器人/车辆/医疗设备/安防/离线环境**必须支持 **纯端侧/边云协同**。技术栈要包含：`ONNX/TensorRT/NCNN/MNN/TVM` 量化部署流水线、`模型蒸馏/剪枝` 自动化、`边缘网关适配`。

### 5.  **商业模式：从「卖项目/卖人天」转向「卖 Seat/卖 Transaction/卖 Outcome/卖 SaaS」**

* **ITSM/HR/客服/外呼/招聘/理赔/尽调** → **按 Seat/月 或 按 Transaction 收费**。
* **投研/合规/图审/造价/研发** → **按 Outcome（件/份/次）或 年度订阅费**。
* **平台/引擎/中台** → **授权费 + 维保费 + 共同开发分成**。
* **拒绝纯「人天外包」**，除非是战略标杆客户（拿数据/拿案例/拿背书）。

## 🛠️ **Botpress 专用：垂直 Agent 「Action 库」通用设计模版 (复制即用)**

为支撑上述 20 个场景快速开发，请在 `src/actions/` 下建立 **领域无关的通用基础 Action 类**，所有垂直 Action 继承复用：

```typescript
// src/actions/base/BaseAction.ts
import { defineAction, z } from '@botpress/sdk'
import { PrismaClient } from '@prisma/client' // 或你的 ORM
import { LLMClient } from '../../tools/llm'
import { VectorClient } from '../../tools/vector'
import { AuditLogger } from '../../tools/audit'

// 统一输入输出规范
export const BaseInputSchema = z.object({
  tenantId: z.string().describe('租户ID，由 Hook 自动注入'),
  userId: z.string().describe('用户ID，由渠道自动注入'),
  conversationId: z.string().describe('会话ID'),
  traceId: z.string().optional().describe('链路追踪ID'),
  // 业务通用字段
  intent: z.string().optional().describe('意图标签'),
  entities: z.record(z.unknown()).optional().describe('实体槽位'),
  context: z.record(z.unknown()).optional().describe('上下文变量'),
})

export const BaseOutputSchema = z.object({
  success: z.boolean(),
  data: z.unknown().optional(),
  message: z.string().optional(), // 给用户看的友好提示
  errorCode: z.string().optional(), // 给系统/坐席看的错误码
  errorDetail: z.string().optional(), // 给开发者看的详细错误
  nextActions: z.array(z.string()).optional(), // 建议后续动作
  auditInfo: z.record(z.unknown()).optional(), // 审计留痕信息
})

export abstract class BaseAction<I extends z.ZodTypeAny, O extends z.ZodTypeAny> {
  protected prisma: PrismaClient
  protected llm: LLMClient
  protected vector: VectorClient
  protected audit: AuditLogger

  constructor(protected readonly ctx: any, protected readonly logger: any) {
    this.prisma = new PrismaClient() // 实际建议用单例/连接池
    this.llm = new LLMClient(ctx)
    this.vector = new VectorClient(ctx)
    this.audit = new AuditLogger(ctx)
  }

  // 子类必须实现：业务逻辑
  protected abstract executeBusiness(input: z.infer<I>): Promise<z.infer<O>>

  // 统一入口：参数校验 -> 审计前 -> 业务执行 -> 审计后 -> 格式化返回
  async execute(input: z.infer<I>): Promise<z.infer<O>> {
    const startTime = Date.now()
    const traceId = input.traceId ?? crypto.randomUUID()
  
    try {
      // 1. 审计前置
      await this.audit.logActionStart({ traceId, actionName: this.constructor.name, input, userId: input.userId, tenantId: input.tenantId })

      // 2. 权限/合规预检 (可选，子类重写)
      await this.preCheck(input)

      // 3. 核心业务执行
      const result = await this.executeBusiness(input)

      // 4. 审计后置
      await this.audit.logActionEnd({ traceId, success: true, output: result, duration: Date.now() - startTime })

      return { success: true, data: result, traceId } as any
    } catch (error: any) {
      // 统一异常处理
      await this.audit.logActionEnd({ traceId, success: false, error: error.message, duration: Date.now() - startTime })
    
      // 区分业务异常 vs 系统异常
      if (error instanceof BusinessError) {
        return { success: false, errorCode: error.code, message: error.message, traceId } as any
      }
    
      this.logger.error({ traceId, error }, 'Action execution failed')
      return { success: false, errorCode: 'INTERNAL_ERROR', message: '系统繁忙，请稍后重试或联系管理员', traceId } as any
    }
  }

  protected async preCheck(input: z.infer<I>) { /* 子类可重写：权限/频控/合规预检 */ }
}

// 业务异常类
export class BusinessError extends Error {
  constructor(public code: string, message: string) {
    super(message)
    this.name = 'BusinessError'
  }
}
```

      amount: z.number(),

      items: z.array(z.object({ sku: z.string(), name: z.string(), qty: z.number(), price: z.number() })),
      buyer: z.object({ email: z.string(), phone: z.string().optional(), address: z.string() }),
      logistics: z.object({ trackingNumber: z.string(), carrier: z.string(), status: z.string(), tracks: z.array(z.object({ time: z.string(), status: z.string(), location: z.string() })) }).optional(),

  async execute({ input, ctx, logger }) {
    // 继承 BaseAction 的能力（审计/LLM/向量库/数据库）
    const action = new QueryOrderAction(ctx, logger)
    return action.execute(input)
  },
})
